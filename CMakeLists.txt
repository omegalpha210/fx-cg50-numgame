cmake_minimum_required(VERSION 3.15)
project(NumGame VERSION 0.1.0 LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
include(GenerateG3A)
find_package(Gint 2.11 REQUIRED)
file(GLOB_RECURSE NUMGAME_SOURCES CONFIGURE_DEPENDS src/*.c)
add_executable(numgame ${NUMGAME_SOURCES})
target_sources(numgame PRIVATE src/power_syscalls.S)
target_include_directories(numgame PRIVATE include src/games)
target_compile_options(numgame PRIVATE -Wall -Wextra -Werror -Wframe-larger-than=2048 -Os -g -fstack-usage)
target_compile_options(numgame PRIVATE $<$<COMPILE_LANGUAGE:C>:-fcallgraph-info=su>)
target_link_options(numgame PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/numgame.map")
target_link_libraries(numgame Gint::Gint)
option(NG_DIAGNOSTIC "Separate diagnostic app and save namespace" OFF)
set(NUMGAME_DIST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/dist" CACHE PATH "Native package output directory")
file(MAKE_DIRECTORY "${NUMGAME_DIST_DIR}")
if(NG_DIAGNOSTIC)
 target_compile_definitions(numgame PRIVATE NG_DIAGNOSTIC)
 target_compile_options(numgame PRIVATE -finstrument-functions)
 set_source_files_properties(src/core/diagnostics.c PROPERTIES COMPILE_OPTIONS -fno-instrument-functions)
 set(NUMGAME_OUTPUT "${NUMGAME_DIST_DIR}/NUMGDIAG.g3a")
 set(NUMGAME_NAME "NUM DIAG")
 set(NUMGAME_INTERNAL "@NGDIAG")
else()
 set(NUMGAME_OUTPUT "${NUMGAME_DIST_DIR}/NUMGAME.g3a")
 set(NUMGAME_NAME "NUM GAME")
 set(NUMGAME_INTERNAL "@NUMGAME")
endif()
generate_g3a(TARGET numgame OUTPUT "${NUMGAME_OUTPUT}" NAME "${NUMGAME_NAME}"
 VERSION "00.02.0000" ICONS assets/icon-uns.png assets/icon-sel.png)
add_custom_command(TARGET numgame POST_BUILD
 COMMAND fxgxa -e "${NUMGAME_OUTPUT}" "--internal=${NUMGAME_INTERNAL}"
 COMMAND fxgxa -r "${NUMGAME_OUTPUT}" VERBATIM)
