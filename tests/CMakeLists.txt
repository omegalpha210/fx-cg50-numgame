cmake_minimum_required(VERSION 3.15)
project(NumGameHost C)
set(CMAKE_C_STANDARD 11)
option(NG_SANITIZE "UBSan" ON)
option(NG_ASAN "ASan (requires working host runtime)" OFF)
option(NG_DIAGNOSTIC "Instrumented diagnostic core and RAM stress" OFF)
add_compile_options(-Wall -Wextra -Werror -g -UNDEBUG)
if(NG_SANITIZE)
 add_compile_options(-fsanitize=undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
 add_link_options(-fsanitize=undefined -fno-sanitize-recover=all)
endif()
if(NG_ASAN)
 add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address)
endif()
include_directories(../include ../src/games)
if(NG_DIAGNOSTIC)
 add_compile_definitions(NG_DIAGNOSTIC)
 add_compile_options(-finstrument-functions)
endif()
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS ../src/*.c)
list(FILTER CORE_SOURCES EXCLUDE REGEX "/main.c$")
add_library(numgame_core ${CORE_SOURCES})
add_executable(export_catalog ../tools/export_catalog.c)
target_link_libraries(export_catalog numgame_core)
add_executable(benchmark_host ../tools/benchmark_host.c)
target_link_libraries(benchmark_host numgame_core)
enable_testing()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/native/run.sh)
 add_test(NAME native_adapter COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/native/run.sh)
endif()
foreach(name integration storage state capture guesscalc_app ui_layout master_integration mastery diagnostics recent entry_beta3 grids_beta3 strategyquick_difficulty)
 if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/test_${name}.c)
  add_executable(test_${name} test_${name}.c)
  target_link_libraries(test_${name} numgame_core)
  add_test(NAME ${name} COMMAND test_${name})
 endif()
endforeach()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/test_grids_extra.c)
 add_executable(test_grids_extra test_grids_extra.c)
 target_compile_definitions(test_grids_extra PRIVATE NG_GRIDS_EXTRA_TEST_MAIN)
 target_link_libraries(test_grids_extra numgame_core)
 add_test(NAME grids_extra COMMAND test_grids_extra)
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/test_guesscalc_extra.c)
 add_executable(test_guesscalc_extra test_guesscalc_extra.c)
 target_link_libraries(test_guesscalc_extra numgame_core)
 add_test(NAME guesscalc_extra COMMAND test_guesscalc_extra)
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/test_strategyquick_extra.c)
 add_executable(test_strategyquick_extra test_strategyquick_extra.c)
 target_compile_definitions(test_strategyquick_extra PRIVATE SQ_EXTRA_TEST_MAIN SQ_EXTRA_APP_TEST)
 target_link_libraries(test_strategyquick_extra numgame_core)
 add_test(NAME strategyquick_extra COMMAND test_strategyquick_extra)
endif()
foreach(name guesscalc grids strategyquick boards)
 if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/test_${name}.c)
  add_executable(test_${name} test_${name}.c)
  target_link_libraries(test_${name} numgame_core)
  if(name STREQUAL "grids")
   target_compile_definitions(test_${name} PRIVATE NG_GRIDS_TEST_MAIN)
  elseif(name STREQUAL "boards")
   target_compile_definitions(test_${name} PRIVATE BOARDS_TEST_MAIN BOARDS_APP_TEST)
  elseif(name STREQUAL "strategyquick")
   target_compile_definitions(test_${name} PRIVATE STRATEGYQUICK_TEST_MAIN)
  endif()
  add_test(NAME ${name} COMMAND test_${name})
 endif()
endforeach()
